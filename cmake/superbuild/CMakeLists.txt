cmake_minimum_required(VERSION 3.17)

# This compiler-free outer project only turns build.c's generated matrix
# manifest into one Ninja graph. Keep matrix policy and workflows in build.c.
project(buster_superbuild NONE)

if (NOT DEFINED BUSTER_SUPERBUILD_MATRIX_FILE OR BUSTER_SUPERBUILD_MATRIX_FILE STREQUAL "")
    message(FATAL_ERROR "BUSTER_SUPERBUILD_MATRIX_FILE is required")
endif()

include("${BUSTER_SUPERBUILD_MATRIX_FILE}")

if (NOT DEFINED BUSTER_SUPERBUILD_SOURCE_DIR OR NOT DEFINED BUSTER_SUPERBUILD_BUILD_DRIVER OR
    NOT DEFINED BUSTER_SUPERBUILD_TREE_COUNT OR NOT DEFINED BUSTER_SUPERBUILD_OUTER_JOBS OR
    NOT DEFINED BUSTER_SUPERBUILD_SELF_HOST_ENABLED OR BUSTER_SUPERBUILD_OUTER_JOBS LESS 1)
    message(FATAL_ERROR "The superbuild matrix file is incomplete")
endif()

if (BUSTER_SUPERBUILD_BUILD_DRIVER STREQUAL "" OR NOT IS_ABSOLUTE "${BUSTER_SUPERBUILD_BUILD_DRIVER}" OR
    NOT EXISTS "${BUSTER_SUPERBUILD_BUILD_DRIVER}" OR IS_DIRECTORY "${BUSTER_SUPERBUILD_BUILD_DRIVER}")
    message(FATAL_ERROR "BUSTER_SUPERBUILD_BUILD_DRIVER must name an existing absolute build driver")
endif()

set_property(GLOBAL PROPERTY JOB_POOLS "buster_matrix_cpu=${BUSTER_SUPERBUILD_OUTER_JOBS}")

add_custom_target(buster_compile)
add_custom_target(buster_matrix ALL)

if (BUSTER_SUPERBUILD_TREE_COUNT LESS 1)
    message(FATAL_ERROR "The superbuild matrix must contain at least one compiler tree")
endif()

if (BUSTER_SUPERBUILD_SELF_HOST_ENABLED)
    if (BUSTER_SUPERBUILD_SELF_HOST_BUILD_DIRECTORY STREQUAL "" OR BUSTER_SUPERBUILD_SELF_HOST_CONFIG STREQUAL "" OR
        BUSTER_SUPERBUILD_SELF_HOST_PROVENANCE_RECORD STREQUAL "" OR BUSTER_SUPERBUILD_BUILD_DRIVER STREQUAL "" OR
        NOT IS_ABSOLUTE "${BUSTER_SUPERBUILD_BUILD_DRIVER}" OR NOT EXISTS "${BUSTER_SUPERBUILD_BUILD_DRIVER}" OR
        IS_DIRECTORY "${BUSTER_SUPERBUILD_BUILD_DRIVER}" OR BUSTER_SUPERBUILD_SELF_HOST_POOL_JOBS LESS 1 OR
        NOT IS_ABSOLUTE "${BUSTER_SUPERBUILD_SELF_HOST_PROVENANCE_RECORD}" OR
        NOT DEFINED BUSTER_SUPERBUILD_SELF_HOST_PRODUCER_CLEAN_REQUIRED OR
        BUSTER_SUPERBUILD_SELF_HOST_PRODUCER_CLEAN_REQUIRED LESS 1 OR
        (NOT BUSTER_SUPERBUILD_SELF_HOST_FUZZ_AVAILABLE STREQUAL "0" AND NOT BUSTER_SUPERBUILD_SELF_HOST_FUZZ_AVAILABLE STREQUAL "1") OR
        BUSTER_SUPERBUILD_SELF_HOST_POOL_JOBS GREATER 1 OR BUSTER_SUPERBUILD_SELF_HOST_DEPENDS_ON_COMPILE LESS 1 OR
        BUSTER_SUPERBUILD_SELF_HOST_USES_INNER_NINJA GREATER 0 OR BUSTER_SUPERBUILD_SELF_HOST_TREE_INDEX LESS 0 OR
        BUSTER_SUPERBUILD_SELF_HOST_TREE_INDEX GREATER_EQUAL BUSTER_SUPERBUILD_TREE_COUNT)
        message(FATAL_ERROR "The superbuild self-host plan is incomplete or unsafe")
    endif()
endif()

math(EXPR BUSTER_SUPERBUILD_LAST_TREE "${BUSTER_SUPERBUILD_TREE_COUNT} - 1")
foreach(BUSTER_SUPERBUILD_TREE_INDEX RANGE 0 ${BUSTER_SUPERBUILD_LAST_TREE})
    set(BUSTER_TREE_PREFIX "BUSTER_SUPERBUILD_TREE_${BUSTER_SUPERBUILD_TREE_INDEX}")
    set(BUSTER_TREE_BUILD_DIRECTORY "${${BUSTER_TREE_PREFIX}_BUILD_DIRECTORY}")
    set(BUSTER_TREE_NATIVE_CONFIG "${${BUSTER_TREE_PREFIX}_NATIVE_CONFIG}")
    set(BUSTER_TREE_BUILD_TARGETS "${${BUSTER_TREE_PREFIX}_BUILD_TARGETS}")
    set(BUSTER_TREE_PARALLEL_JOBS "${${BUSTER_TREE_PREFIX}_PARALLEL_JOBS}")
    set(BUSTER_TREE_UNITY_ONLY "${${BUSTER_TREE_PREFIX}_UNITY_ONLY}")
    set(BUSTER_TREE_TEST_0_CONFIG "${${BUSTER_TREE_PREFIX}_TEST_0_CONFIG}")
    set(BUSTER_TREE_TEST_0_TARGET "${${BUSTER_TREE_PREFIX}_TEST_0_TARGET}")
    set(BUSTER_TREE_TEST_1_CONFIG "${${BUSTER_TREE_PREFIX}_TEST_1_CONFIG}")
    set(BUSTER_TREE_TEST_1_TARGET "${${BUSTER_TREE_PREFIX}_TEST_1_TARGET}")
    set(BUSTER_TREE_ANALYZE_CONFIG "${${BUSTER_TREE_PREFIX}_ANALYZE_CONFIG}")

    if (BUSTER_TREE_BUILD_DIRECTORY STREQUAL "" OR BUSTER_TREE_NATIVE_CONFIG STREQUAL "" OR
        BUSTER_TREE_BUILD_TARGETS STREQUAL "" OR BUSTER_TREE_PARALLEL_JOBS LESS 1 OR
        (NOT BUSTER_TREE_UNITY_ONLY STREQUAL "0" AND NOT BUSTER_TREE_UNITY_ONLY STREQUAL "1") OR
        BUSTER_TREE_TEST_0_CONFIG STREQUAL "" OR BUSTER_TREE_TEST_0_TARGET STREQUAL "")
        message(FATAL_ERROR "Compiler tree ${BUSTER_SUPERBUILD_TREE_INDEX} is incomplete")
    endif()

    set(BUSTER_TREE_BUILD_OPTIONS)
    if (BUSTER_SUPERBUILD_VERBOSE)
        list(APPEND BUSTER_TREE_BUILD_OPTIONS --verbose)
    endif()
    if (BUSTER_SUPERBUILD_QUIET)
        list(APPEND BUSTER_TREE_BUILD_OPTIONS -- --quiet)
    endif()

    set(BUSTER_TREE_COMPILE_TARGET "buster_compile_${BUSTER_SUPERBUILD_TREE_INDEX}")
    add_custom_target(${BUSTER_TREE_COMPILE_TARGET}
        COMMAND "${CMAKE_COMMAND}" --build "${BUSTER_TREE_BUILD_DIRECTORY}"
            --config "${BUSTER_TREE_NATIVE_CONFIG}"
            --target ${BUSTER_TREE_BUILD_TARGETS}
            --parallel "${BUSTER_TREE_PARALLEL_JOBS}"
            ${BUSTER_TREE_BUILD_OPTIONS}
        WORKING_DIRECTORY "${BUSTER_SUPERBUILD_SOURCE_DIR}"
        JOB_POOL buster_matrix_cpu
        VERBATIM
    )
    add_dependencies(buster_compile ${BUSTER_TREE_COMPILE_TARGET})

    set(BUSTER_TREE_TEST_COMMANDS
        COMMAND "${CMAKE_COMMAND}" -E env
            "BUSTER_TEST_JOBS=${BUSTER_TREE_PARALLEL_JOBS}"
            "${CMAKE_COMMAND}" --build "${BUSTER_TREE_BUILD_DIRECTORY}"
            --config "${BUSTER_TREE_TEST_0_CONFIG}"
            --target "${BUSTER_TREE_TEST_0_TARGET}"
            ${BUSTER_TREE_BUILD_OPTIONS}
    )
    if (NOT BUSTER_TREE_TEST_1_CONFIG STREQUAL "")
        if (BUSTER_TREE_TEST_1_TARGET STREQUAL "")
            message(FATAL_ERROR "Compiler tree ${BUSTER_SUPERBUILD_TREE_INDEX} has an incomplete second test")
        endif()
        list(APPEND BUSTER_TREE_TEST_COMMANDS
            COMMAND "${CMAKE_COMMAND}" -E env
                "BUSTER_TEST_JOBS=${BUSTER_TREE_PARALLEL_JOBS}"
                "${CMAKE_COMMAND}" --build "${BUSTER_TREE_BUILD_DIRECTORY}"
                --config "${BUSTER_TREE_TEST_1_CONFIG}"
                --target "${BUSTER_TREE_TEST_1_TARGET}"
                ${BUSTER_TREE_BUILD_OPTIONS}
        )
    endif()
    if (NOT BUSTER_TREE_ANALYZE_CONFIG STREQUAL "")
        list(APPEND BUSTER_TREE_TEST_COMMANDS
            COMMAND "${CMAKE_COMMAND}" --build "${BUSTER_TREE_BUILD_DIRECTORY}"
                --config "${BUSTER_TREE_ANALYZE_CONFIG}"
                --target clang_analyze
                ${BUSTER_TREE_BUILD_OPTIONS}
        )
    endif()

    set(BUSTER_TREE_TEST_TARGET "buster_test_${BUSTER_SUPERBUILD_TREE_INDEX}")
    add_custom_target(${BUSTER_TREE_TEST_TARGET}
        ${BUSTER_TREE_TEST_COMMANDS}
        WORKING_DIRECTORY "${BUSTER_SUPERBUILD_SOURCE_DIR}"
        JOB_POOL buster_matrix_cpu
        VERBATIM
    )
    # Preserve tree-level pipelining: a completed tree can release its
    # compiler slot to its test command while other trees continue building.
    # The shared pool still bounds both kinds of command globally.
    add_dependencies(${BUSTER_TREE_TEST_TARGET} ${BUSTER_TREE_COMPILE_TARGET})
    add_dependencies(buster_matrix ${BUSTER_TREE_TEST_TARGET})
endforeach()

if (BUSTER_SUPERBUILD_SELF_HOST_ENABLED)
    set(BUSTER_SELF_HOST_COMPILE_TARGET "buster_compile_${BUSTER_SUPERBUILD_SELF_HOST_TREE_INDEX}")
    if (NOT TARGET ${BUSTER_SELF_HOST_COMPILE_TARGET})
        message(FATAL_ERROR "The superbuild self-host dependency tree does not exist")
    endif()
    set(BUSTER_SELF_HOST_TREE_PREFIX "BUSTER_SUPERBUILD_TREE_${BUSTER_SUPERBUILD_SELF_HOST_TREE_INDEX}")
    set(BUSTER_SELF_HOST_TREE_BUILD_DIRECTORY "${${BUSTER_SELF_HOST_TREE_PREFIX}_BUILD_DIRECTORY}")
    if (NOT "${BUSTER_SELF_HOST_TREE_BUILD_DIRECTORY}" STREQUAL "${BUSTER_SUPERBUILD_SELF_HOST_BUILD_DIRECTORY}" OR
        NOT "${BUSTER_SUPERBUILD_SELF_HOST_CONFIG}" STREQUAL "Release")
        message(FATAL_ERROR "The superbuild self-host producer does not match its compile dependency")
    endif()

    set(BUSTER_SELF_HOST_COMMAND_OPTIONS
        --build-directory "${BUSTER_SUPERBUILD_SELF_HOST_BUILD_DIRECTORY}"
        --config "${BUSTER_SUPERBUILD_SELF_HOST_CONFIG}"
        --provenance-record "${BUSTER_SUPERBUILD_SELF_HOST_PROVENANCE_RECORD}"
    )
    if (BUSTER_SUPERBUILD_SELF_HOST_CI)
        list(APPEND BUSTER_SELF_HOST_COMMAND_OPTIONS --ci)
    endif()
    if (BUSTER_SUPERBUILD_SELF_HOST_FUZZ_AVAILABLE)
        list(APPEND BUSTER_SELF_HOST_COMMAND_OPTIONS --fuzz)
    else()
        list(APPEND BUSTER_SELF_HOST_COMMAND_OPTIONS --no-fuzz)
    endif()

    add_custom_target(buster_self_host
        COMMAND "${BUSTER_SUPERBUILD_BUILD_DRIVER}" self_host_from_existing ${BUSTER_SELF_HOST_COMMAND_OPTIONS}
        WORKING_DIRECTORY "${BUSTER_SUPERBUILD_SOURCE_DIR}"
        JOB_POOL buster_matrix_cpu
        VERBATIM
    )
    # The worker consumes one slot from the same pool and runs direct compiler
    # processes; it never starts an inner Ninja process. Waiting only for the
    # producer compile target lets it overlap the remaining matrix tail.
    add_dependencies(buster_self_host ${BUSTER_SELF_HOST_COMPILE_TARGET})
    add_dependencies(buster_matrix buster_self_host)
endif()

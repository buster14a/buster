0005844: BUSTER_GLOBAL_LOCAL bool codegen_canonical_a64_memory_operation_base(CodegenBuffer* buffer, u32 register_number, u32 offset, u32 size, bool store,
0005845:                                                                      bool sign_extend, u32 base_register)
0005846: {
0005847:     u32 scale = size == 8 ? 8 : size == 4 ? 4 : size == 2 ? 2 : size == 1 ? 1 : 0;
0005848:     if (!scale || register_number > 31)
0005849:     {
0005850:         return false;
0005851:     }
0005852:     // The scaled unsigned-immediate form addresses only multiples of the
0005853:     // access width, and a bit-field's storage unit lands wherever packing put
0005854:     // it: `struct __attribute__((packed)) { char a; int v : 32; }` is read
0005855:     // through the four bytes at offset one, and the pieces of a field with no
0005856:     // single unit land at every offset in turn. The unscaled form -- LDUR and
0005857:     // STUR -- takes any byte offset in a nine-bit field, which is exactly the
0005858:     // offsets the scaled one cannot reach. Beyond that field the address is
0005859:     // materialized, and then the offset is zero and scaled again.
0005860:     bool unscaled = offset % scale != 0;
0005861:     bool indirect = unscaled ? offset > A64_UNSCALED_IMM_MAX : offset / scale > A64_IMM12_MAX;
0005862:     if (indirect)
0005863:     {
0005864:         codegen_canonical_a64_base_address(buffer, 16, base_register, offset);
0005865:         offset = 0;
0005866:         unscaled = false;
0005867:     }
0005868:     if (unscaled)
0005869:     {
0005870:         u32 unscaled_instruction = store         ? (size == 8 ? 0xf80003e0 : size == 4 ? 0xb80003e0 : size == 2 ? 0x780003e0 : 0x380003e0)
0005871:                                    : sign_extend ? (size == 4 ? 0xb88003e0 : size == 2 ? 0x788003e0 : size == 1 ? 0x388003e0 : 0xf84003e0)
0005872:                                                  : (size == 8 ? 0xf84003e0 : size == 4 ? 0xb84003e0 : size == 2 ? 0x784003e0 : 0x384003e0);
0005873:         codegen_emit_u32(buffer, (unscaled_instruction & ~(31u << 5)) | (offset << 12) | (base_register << 5) | register_number);
0005874:         return true;
0005875:     }
0005876:     u32 instruction = 0;
0005877:     if (store)
0005878:     {
0005879:         instruction = size == 8 ? 0xf90003e0 : size == 4 ? 0xb90003e0 : size == 2 ? 0x790003e0 : 0x390003e0;
0005880:     }
0005881:     else if (sign_extend)
0005882:     {
0005883:         instruction = size == 4 ? 0xb98003e0 : size == 2 ? 0x798003e0 : size == 1 ? 0x398003e0 : 0xf94003e0;
0005884:     }
0005885:     else
0005886:     {
0005887:         instruction = size == 8 ? 0xf94003e0 : size == 4 ? 0xb94003e0 : size == 2 ? 0x794003e0 : 0x394003e0;
0005888:     }
0005889:     codegen_emit_u32(buffer, (instruction & ~(31u << 5)) | ((offset / scale) << 10) | ((indirect ? 16u : base_register) << 5) | register_number);
0005890:     return true;
0005891: }

0005932: // The address of an outgoing-argument slot. The stack pointer is only sixteen
0005933: // aligned through the body, so a slot an argument needs more alignment than
0005934: // that is reserved with room to spare and rounded up here, the same lea/add/and

0005937: {

0005939:         (alignment > CODEGEN_X64_STACK_ALIGNMENT && (alignment > INT32_MAX || (alignment & (alignment - 1)))))
0005940:     {
0005941:         return false;

0005957:         };

0005959:             codegen_canonical_x64_metadata_gpr(register_index, 64),

0005985: // The caller-owned copy of one indirectly passed argument, moved from its frame
0005986: // slot into the outgoing area an eightbyte at a time. An argument that wants
0005987: // more than the stack pointer's sixteen bytes of alignment is written through
0005988: // r11 instead, holding the rounded-up address of its over-reserved slot: r11 is
0005989: // volatile, carries no argument, and every copy re-materializes it.
0005990: BUSTER_GLOBAL_LOCAL bool codegen_canonical_x64_copy_frame_to_rsp(CodegenBuffer* buffer, u32 source_offset, u32 frame_base_offset,
0005991:                                                                  u32 destination_offset, u32 size, u32 alignment)
0005992: {
0005993:     if (!buffer || !size || source_offset > INT32_MAX || destination_offset > INT32_MAX)
0005994:     {
0005995:         return false;
0005996:     }
0005997:     bool through_scratch = alignment > CODEGEN_X64_STACK_ALIGNMENT;
0005998:     if (through_scratch)
0005999:     {
0006000:         if (!codegen_canonical_x64_rsp_address(buffer, X64_REGISTER_R11, destination_offset, alignment))
0006001:         {
0006002:             return false;
0006003:         }
0006004:         destination_offset = 0;
0006005:     }
0006006:     u64 copied = 0;
0006007:     while (copied < size)
0006008:     {
0006009:         u64 source_offset_within_value = (u64)source_offset + copied;
0006010:         u64 destination = (u64)destination_offset + copied;
0006011:         s32 source = codegen_canonical_x64_rebase_frame_displacement(buffer, -(s64)source_offset + (s64)copied, frame_base_offset);
0006012:         u32 chunk = codegen_canonical_copy_chunk((u64)size - copied, source_offset_within_value, destination);
0006013:         if (source_offset_within_value > INT32_MAX || destination > INT32_MAX || buffer->error != CODEGEN_ERROR_NONE)
0006014:         {
0006015:             return false;
0006016:         }
0006017:         u16 width = (u16)(chunk * 8);
0006018:         BusterX86MetadataPhysicalOperand load_operands[2] = {
0006019:             codegen_canonical_x64_metadata_gpr(X64_REGISTER_RAX, chunk <= 2 ? 32 : width),
0006020:             codegen_canonical_x64_metadata_memory(X64_REGISTER_RBP, width, source),
0006021:         };
0006022:         String8 load_mnemonic = chunk <= 2 ? S8("MOVZX") : S8("MOV");
0006023:         if (!codegen_canonical_x64_metadata_emit(buffer, load_mnemonic, load_operands, BUSTER_ARRAY_LENGTH(load_operands)))
0006024:         {
0006025:             return false;
0006026:         }
0006027:         BusterX86MetadataPhysicalOperand store_operands[2] = {
0006028:             codegen_canonical_x64_metadata_memory(through_scratch ? X64_REGISTER_R11 : X64_REGISTER_RSP, width, (s64)destination),
0006029:             codegen_canonical_x64_metadata_gpr(X64_REGISTER_RAX, width),
0006030:         };
0006031:         if (!codegen_canonical_x64_metadata_emit(buffer, S8("MOV"), store_operands, BUSTER_ARRAY_LENGTH(store_operands)))
0006032:         {
0006033:             return false;
0006034:         }
0006035:         copied += chunk;
0006036:     }
0006037:     return true;
0006038: }

0006087: BUSTER_GLOBAL_LOCAL bool codegen_canonical_a64_float_memory_operation_base(CodegenBuffer* buffer, u32 register_number, u32 offset, u32 size, bool store,
0006088:                                                                            u32 base_register)
0006089: {
0006090:     // The b and h forms carry the one- and two-byte short vectors AAPCS64
0006091:     // hands the low bytes of a V register. Without them the HFA capture
0006092:     // path — whose part classification already routes a vector here —
0006093:     // emitted nothing at all and left the parameter's slot zero, since
0006094:     // its callers do not check this helper's result.
0006095:     u32 scale = size == 16 ? 16 : size == 8 ? 8 : size == 4 ? 4 : size == 2 ? 2 : size == 1 ? 1 : 0;
0006096:     if (!scale || offset % scale || register_number > 31)
0006097:     {
0006098:         return false;
0006099:     }
0006100:     bool indirect = offset / scale > A64_IMM12_MAX;
0006101:     if (indirect)
0006102:     {
0006103:         codegen_canonical_a64_base_address(buffer, 16, base_register, offset);
0006104:         offset = 0;
0006105:     }
0006106:     u32 instruction = store ? (size == 16   ? 0x3d8003e0
0006107:                                : size == 8  ? 0xfd0003e0
0006108:                                : size == 4  ? 0xbd0003e0
0006109:                                : size == 2  ? 0x7d0003e0
0006110:                                             : 0x3d0003e0)
0006111:                             : (size == 16   ? 0x3dc003e0
0006112:                                : size == 8  ? 0xfd4003e0
0006113:                                : size == 4  ? 0xbd4003e0
0006114:                                : size == 2  ? 0x7d4003e0
0006115:                                             : 0x3d4003e0);
0006116:     codegen_emit_u32(buffer, (instruction & ~(31u << 5)) | ((offset / scale) << 10) | ((indirect ? 16u : base_register) << 5) | register_number);
0006117:     return true;
0006118: }

0006481: // Branches to parameterized blocks pass through an out-of-line edge thunk.
0006482: // First capture every source into the frame tile, then publish destinations;
0006483: // no edge can overwrite a source needed by another assignment. Label-address
0006484: // relocations bypass thunks so address-taken block identity never changes.
0006485: BUSTER_GLOBAL_LOCAL u32 c_canonical_edge_thunk(CCanonicalEmitter* emitter, IrProgram* program, IrFunction* function,
0006486:                                               CCanonicalBranchPatch patch, u32 scratch_offset, u32 target_offset)
0006487: {

0006490:     IrPublishedCfg const* cfg = function->published_cfg;

0006492:     if (!edge || !cfg || patch.target.value >= cfg->block_count)
0006493:     {
0006494:         buffer->error = CODEGEN_ERROR_INVALID_IR;
0006495:     }

0006497:     for (u32 pass = 0; pass < 2 && !buffer->error; pass += 1)
0006498:     {

0006500:         for (u32 index = 0; index < target->parameter_count && !buffer->error; index += 1)
0006501:         {

0006503:             IrValueId incoming = cfg->arguments[edge->argument_offset + index];

0006506:                 u64 bytes = (BUSTER_MAX(type->layout.size, (u64)8) + 7) & ~(u64)7;

0006514:                         if (!codegen_canonical_a64_frame_memory_operation(buffer, 9, pass ? temp_offset : value_offset, 8, false, false) ||

0006521:                         }

0006523:                     else
0006524:                     {

0006529:                     }

0006532:             }
0006533:         }
0006534:     }

0006695: // Convert X9 between the compact frame image and the widened D0 ABI image.
0006696: // X10 is the accumulating result and X11 is one extracted lane.
0006697: BUSTER_GLOBAL_LOCAL void codegen_canonical_a64_pack_vector_result(CodegenBuffer* buffer, u32 lane_count, u32 element_bits, bool compact)
0006698: {
0006699:     u32 abi_lane_bits = 64u / lane_count;
0006700:     u32 source_lane_bits = compact ? abi_lane_bits : element_bits;
0006701:     u32 result_lane_bits = compact ? element_bits : abi_lane_bits;
0006702:     codegen_emit_u32(buffer, UINT32_C(0xaa1f03ea)); // mov x10, xzr
0006703:     for (u32 lane = 0; lane < lane_count; lane += 1)
0006704:     {
0006705:         u32 source_offset = lane * source_lane_bits;
0006706:         codegen_emit_u32(buffer, UINT32_C(0xd3400000) | (source_offset << 16) | ((source_offset + element_bits - 1u) << 10) |
0006707:                                      (9u << 5) | 11u); // ubfx x11, x9, #source_offset, #element_bits
0006708:         u32 result_offset = lane * result_lane_bits;
0006709:         if (result_offset)
0006710:         {
0006711:             codegen_emit_u32(buffer, UINT32_C(0xd3400000) | ((64u - result_offset) << 16) | ((63u - result_offset) << 10) |
0006712:                                          (11u << 5) | 11u); // lsl x11, x11, #result_offset
0006713:         }
0006714:         codegen_emit_u32(buffer, UINT32_C(0xaa0b014a)); // orr x10, x10, x11
0006715:     }

0006717: }

0006719: // Patches one local AArch64 branch after its target has been emitted. The
0006720: // canonical emitter uses this for the fixed-shape loops inside one IR row;
0006721: // ordinary IR block branches continue through CCanonicalBranchPatch below.
0006722: BUSTER_GLOBAL_LOCAL bool codegen_canonical_a64_patch_local_branch(CodegenBuffer* buffer, u32 branch_offset, u32 target_offset, bool conditional)
0006723: {
0006724:     bool result = false;
0006725:     if (buffer && buffer->error == CODEGEN_ERROR_NONE && branch_offset <= UINT32_MAX - 4 && branch_offset + 4 <= buffer->count && target_offset <= UINT32_MAX)
0006726:     {
0006727:         s64 delta = (s64)target_offset - (s64)branch_offset;
0006728:         s64 minimum = conditional ? -(s64)(1 << 20) : -(s64)(1 << 27);
0006729:         s64 maximum = conditional ? (s64)(1 << 20) : (s64)(1 << 27);
0006730:         if ((delta & 3) || delta < minimum || delta >= maximum)
0006731:         {
0006732:             buffer->error = CODEGEN_ERROR_CAPACITY;
0006733:         }
0006734:         else
0006735:         {
0006736:             u32 instruction = 0;
0006737:             memcpy(&instruction, buffer->bytes + branch_offset, sizeof(instruction));
0006738:             u32 immediate_mask = conditional ? UINT32_C(0x00ffffe0) : UINT32_C(0x03ffffff);
0006739:             if (instruction & immediate_mask)
0006740:             {
0006741:                 // A local patch must start from the zero-immediate template;
0006742:                 // silently OR-ing onto an old displacement would corrupt the
0006743:                 // loop target while still producing a decodable branch.
0006744:                 buffer->error = CODEGEN_ERROR_INVALID_IR;
0006745:             }
0006746:             else
0006747:             {
0006748:                 u32 immediate = (u32)(delta >> 2);
0006749:                 instruction |= conditional ? (immediate & 0x7ffff) << 5 : immediate & 0x03ffffff;
0006750:                 memcpy(buffer->bytes + branch_offset, &instruction, sizeof(instruction));
0006751:                 result = true;
0006752:             }
0006753:         }
0006754:     }
0006755:     else if (buffer && buffer->error == CODEGEN_ERROR_NONE)
0006756:     {
0006757:         buffer->error = CODEGEN_ERROR_CAPACITY;
0006758:     }
0006759:     return result;
0006760: }

0006762: // Emits the canonical 128-bit divide/remainder loop. Inputs arrive in
0006763: // x9:x10 (dividend) and x11:x12 (divisor), low eightbyte first. The loop
0006764: // shifts one dividend bit into x13:x14, subtracts the divisor when the
0006765: // partial remainder is large enough, and accumulates the quotient in x0:x1.
0006766: // Signed operations normalize both operands before the loop and restore the
0006767: // quotient/remainder sign afterwards; unsigned operations use the same loop
0006768: // without that normalization. On success the selected result is x9:x10.
0006769: BUSTER_GLOBAL_LOCAL bool codegen_canonical_a64_i128_divide(CodegenBuffer* buffer, bool signed_division, bool remainder_result)
0006770: {
0006771:     bool result = false;
0006772:     if (buffer)
0006773:     {
0006774:         if (signed_division)
0006775:         {
0006776:             // x15/x16 retain the operand sign masks; the pair XOR/subtract
0006777:             // turns each negative operand into its unsigned magnitude.
0006778:             codegen_emit_u32(buffer, 0x937ffd4f); // asr x15, x10, #63
0006779:             codegen_emit_u32(buffer, 0xca0f0129); // eor x9, x9, x15
0006780:             codegen_emit_u32(buffer, 0xca0f014a); // eor x10, x10, x15
0006781:             codegen_emit_u32(buffer, 0xeb0f0129); // subs x9, x9, x15
0006782:             codegen_emit_u32(buffer, 0xda0f014a); // sbc x10, x10, x15
0006783:             codegen_emit_u32(buffer, 0x937ffd90); // asr x16, x12, #63
0006784:             codegen_emit_u32(buffer, 0xca10016b); // eor x11, x11, x16

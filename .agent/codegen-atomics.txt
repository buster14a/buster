  1374: // executes them -- a data16 on a 64-bit lea and two on a rex.W call change
  1375: // nothing -- and they are there so the sixteen bytes have a shape a linker
  1376: // can recognize: relaxing general-dynamic to initial-exec or local-exec
  1377: // replaces all sixteen, and both `ld` and this tree's own linker match on
  1378: // exactly these bytes to do it.  That makes them data whose identity is the
  1379: // contract, which no metadata form can express, since the encoder's job is to
  1380: // choose the shortest encoding of an instruction and the shortest one here is
  1381: // the wrong answer.  The site is registered as a neutral fixed sequence in
  1382: // `machine_x86_64_neutral_patch_sites` so it stays reviewable rather than
  1383: // becoming an unaudited byte writer.
  1384: BUSTER_GLOBAL_LOCAL void codegen_canonical_x64_thread_local_general_dynamic(CodegenBuffer* buffer, u32* address_offset, u32* helper_offset)
  1385: {
  1386:     codegen_emit_u8(buffer, 0x66);
  1387:     codegen_emit_u8(buffer, 0x48);
  1388:     codegen_emit_u8(buffer, 0x8d);
  1389:     codegen_emit_u8(buffer, 0x3d);
  1390:     *address_offset = (u32)buffer->count;
  1391:     codegen_emit_u32(buffer, 0);
  1392:     codegen_emit_u8(buffer, 0x66);
  1393:     codegen_emit_u8(buffer, 0x66);
  1394:     codegen_emit_u8(buffer, 0x48);
  1395:     codegen_emit_u8(buffer, 0xe8);
  1396:     *helper_offset = (u32)buffer->count;
  1397:     codegen_emit_u32(buffer, 0);
  1398: }

  1130: BUSTER_GLOBAL_LOCAL bool codegen_inline_assembly_constraint_shape_valid(u64 constraint, u32 operand_index, u32 operand_count, u32* match_index_out)
  1131: {
  1132:     if ((constraint & ~IR_INLINE_ASSEMBLY_CONSTRAINT_KNOWN_MASK) ||
  1133:         (constraint & IR_INLINE_ASSEMBLY_CONSTRAINT_CLASS_MASK) >= IR_INLINE_ASSEMBLY_CONSTRAINT_COUNT)
  1134:     {
  1135:         return false;
  1136:     }

  1138:     bool read_write = (constraint & IR_INLINE_ASSEMBLY_CONSTRAINT_READ_WRITE) != 0;
  1139:     bool matching = (constraint & IR_INLINE_ASSEMBLY_CONSTRAINT_MATCH) != 0;
  1140:     if ((read_write && !output) || (matching && (output || read_write)))
  1141:     {
  1142:         return false;

  1151:         return false;
  1152:     }
  1153:     if (match_index_out)
  1154:     {
  1155:         *match_index_out = match_index;
  1156:     }
  1157:     return true;

  1223: BUSTER_GLOBAL_LOCAL u32 codegen_canonical_a64_nop_count(String8 literal)
  1224: {
  1225:     u32 count = 0;
  1226:     u64 offset = 0;
  1227:     while (offset < literal.length)
  1228:     {
  1229:         while (offset < literal.length && (literal.pointer[offset] == ' ' || literal.pointer[offset] == '\t' || literal.pointer[offset] == '\r' ||
  1230:                                             literal.pointer[offset] == '\n'))
  1231:         {
  1232:             offset += 1;
  1233:         }
  1234:         while (offset + 1 < literal.length && literal.pointer[offset] == '\\' &&
  1235:                (literal.pointer[offset + 1] == 'n' || literal.pointer[offset + 1] == 'r' || literal.pointer[offset + 1] == 't'))
  1236:         {
  1237:             offset += 2;
  1238:             while (offset < literal.length && (literal.pointer[offset] == ' ' || literal.pointer[offset] == '\t' || literal.pointer[offset] == '\r' ||
  1239:                                                 literal.pointer[offset] == '\n'))
  1240:             {
  1241:                 offset += 1;
  1242:             }
  1243:         }
  1244:         if (offset == literal.length || literal.pointer[offset] == 0)
  1245:         {
  1246:             break;
  1247:         }
  1248:         if (offset + 3 > literal.length || literal.pointer[offset] != 'n' || literal.pointer[offset + 1] != 'o' || literal.pointer[offset + 2] != 'p')
  1249:         {
  1250:             return 0;
  1251:         }
  1252:         offset += 3;
  1253:         if (count == UINT32_MAX)
  1254:         {
  1255:             return 0;
  1256:         }
  1257:         count += 1;
  1258:     }

  1368: // The ELF general-dynamic thread-local pair, sixteen fixed bytes:
  1369: //
  1370: //   66 48 8d 3d <r32>  data16 lea rdi, [rip + sym@TLSGD]
  1371: //   66 66 48 e8 <r32>  data16 data16 rex.W call __tls_get_addr
  1372: //
  1373: // The prefixes are the sequence rather than an encoding of it.  Nothing
  1374: // executes them -- a data16 on a 64-bit lea and two on a rex.W call change
  1375: // nothing -- and they are there so the sixteen bytes have a shape a linker
  1376: // can recognize: relaxing general-dynamic to initial-exec or local-exec
  1377: // replaces all sixteen, and both `ld` and this tree's own linker match on
  1378: // exactly these bytes to do it.  That makes them data whose identity is the
  1379: // contract, which no metadata form can express, since the encoder's job is to
  1380: // choose the shortest encoding of an instruction and the shortest one here is
  1381: // the wrong answer.  The site is registered as a neutral fixed sequence in
  1382: // `machine_x86_64_neutral_patch_sites` so it stays reviewable rather than
  1383: // becoming an unaudited byte writer.
  1384: BUSTER_GLOBAL_LOCAL void codegen_canonical_x64_thread_local_general_dynamic(CodegenBuffer* buffer, u32* address_offset, u32* helper_offset)
  1385: {
  1386:     codegen_emit_u8(buffer, 0x66);
  1387:     codegen_emit_u8(buffer, 0x48);
  1388:     codegen_emit_u8(buffer, 0x8d);
  1389:     codegen_emit_u8(buffer, 0x3d);
  1390:     *address_offset = (u32)buffer->count;
  1391:     codegen_emit_u32(buffer, 0);
  1392:     codegen_emit_u8(buffer, 0x66);
  1393:     codegen_emit_u8(buffer, 0x66);
  1394:     codegen_emit_u8(buffer, 0x48);
  1395:     codegen_emit_u8(buffer, 0xe8);
  1396:     *helper_offset = (u32)buffer->count;
  1397:     codegen_emit_u32(buffer, 0);
  1398: }

cmake_minimum_required(VERSION 3.25)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "" FORCE)
set(CMAKE_C_FLAGS_RELEASE "" CACHE STRING "" FORCE)
project(buster_codegen_fixture LANGUAGES C)
option(BUSTER_FIXTURE_SPLIT "Exercise the production optimized-IR partition helpers" ON)
option(BUSTER_FIXTURE_FAIL_SPLIT "Inject a failed splitter into the dependency fixture" OFF)
set(BUSTER_FIXTURE_SANITIZER "" CACHE STRING "address or memory: every unit and the link instrumented, as a sanitizer build's")
if(NOT DEFINED BUSTER_CODEGEN_MODULE_DIR)
    get_filename_component(BUSTER_CODEGEN_MODULE_DIR ../../../cmake ABSOLUTE)
endif()
set(buster_parallel_codegen ${BUSTER_FIXTURE_SPLIT})
# split.c's units, as a sanitizer build's Release splits them (BusterCodegen.cmake's buster_codegen_split_unit); every
# build's Release, sanitizer or not, used to.
set(buster_split_parts ${BUSTER_FIXTURE_SPLIT})
set(buster_arch x86-64)
# A sanitizer build's flags (BusterSettings.cmake's), which the parts' code generation takes too: AddressSanitizer's
# comdat constructors and aliases, MemorySanitizer's retained constructor and PIE.
set(buster_sanitizer_flags "")
set(link_mode -no-pie)
if(BUSTER_FIXTURE_SANITIZER STREQUAL "address")
    set(buster_sanitizer_flags -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
elseif(BUSTER_FIXTURE_SANITIZER STREQUAL "memory")
    set(buster_sanitizer_flags -fsanitize=memory -fsanitize-memory-track-origins=2 -fno-sanitize-recover=all -fno-omit-frame-pointer -fPIE)
    set(link_mode -pie)
elseif(NOT BUSTER_FIXTURE_SANITIZER STREQUAL "")
    message(FATAL_ERROR "codegen fixture: BUSTER_FIXTURE_SANITIZER must be address, memory or empty")
endif()
get_filename_component(compiler_file "${CMAKE_C_COMPILER}" REALPATH)
include("${BUSTER_CODEGEN_MODULE_DIR}/BusterCodegen.cmake")
if(buster_parallel_codegen)
    buster_codegen_tools(available missing)
    if(NOT available)
        message(FATAL_ERROR "codegen fixture: parallel native generation needs ${missing}")
    endif()
    buster_codegen_splitter()
    if(BUSTER_FIXTURE_FAIL_SPLIT)
        set(BUSTER_CODEGEN_SPLIT /bin/false)
    endif()
endif()
foreach(name checker backend)
    set(unit main_unit_${name})
    add_library(buster_${unit} OBJECT ${name}_test.c)
    target_compile_options(buster_${unit} PRIVATE -march=x86-64 -fwrapv -funsigned-char -fno-strict-aliasing
        -fno-stack-protector -Wall -Wextra -Werror -O3 ${buster_sanitizer_flags})
    if(buster_parallel_codegen)
        target_compile_options(buster_${unit} PRIVATE -emit-llvm)
        buster_codegen_unit(${unit})
    endif()
    buster_native_object(object ${unit})
    list(APPEND objects "${object}")
endforeach()
add_executable(codegen_fixture main_test.c ${objects})
target_compile_options(codegen_fixture PRIVATE -Wall -Wextra -Werror -O3 -fno-stack-protector ${buster_sanitizer_flags})
target_link_options(codegen_fixture PRIVATE ${link_mode} ${buster_sanitizer_flags})
target_link_libraries(codegen_fixture PRIVATE pthread)
enable_testing()
add_test(NAME codegen-bindings COMMAND codegen_fixture)

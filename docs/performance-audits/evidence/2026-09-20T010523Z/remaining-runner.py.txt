def sample(case, leg, name, round_, pair, measured):
    global expected, summary
    command=['taskset','-c','2',str(root/name/'build/Release/ide'),*args]
    start=time.monotonic_ns()
    p=subprocess.run(command,cwd=source,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    elapsed=time.monotonic_ns()-start
    prefix=f'{case}-{round_}-{pair}-{leg}'
    (root/(prefix+'.log')).write_bytes(p.stdout)
    if p.returncode: raise RuntimeError(prefix+' failed')
    sha=digest(output)
    lines=[s for s in p.stdout.decode().splitlines() if s.startswith(('IR_FAST ','IR_LOCAL_PROMOTION ','IR_LOCAL_PROMOTION_WORK ','CODEGEN '))]
    if expected is None: expected,summary=sha,lines
    if sha != expected or lines != summary: raise RuntimeError(prefix+' output mismatch')
    records.append(dict(case=case,leg=leg,compiler=name,round=round_,pair=pair,measured=measured,wall_ns=elapsed,sha256=sha))
    (root/'samples.json').write_text(json.dumps({'command':args,'identities':identities,'last_case_summary':summary,'records':records},indent=2))

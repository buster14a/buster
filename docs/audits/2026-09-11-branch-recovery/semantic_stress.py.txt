#!/usr/bin/env python3
"""Generate semantic stress inputs and compare two compilers on identical files.

All generated programs are compiled, never executed. The default syntax-only
path includes Buster's canonical-IR lowering. --objects additionally checks
byte-identical object output. Measurements alternate compiler order, include a
warmup, retain every sample, and never make wall time a CI threshold.

Example (run from the repository root):
  python3 tools/semantic_stress.py --baseline /tmp/ide-old --candidate build/Release/ide --sizes 512,2048,8192 --repetitions 7
"""
import argparse
import hashlib
import json
import os
from pathlib import Path
import platform
import signal
import statistics
import subprocess
import tempfile
import time

try:
    import resource
except ImportError:
    resource = None

KINDS = ("loops", "deep", "symbols", "constants", "generic", "diagnostics")


def source(kind, count):
    """Return source and the required exit status; count is a positive integer."""
    if count < 1:
        raise ValueError("counts must be positive")
    if kind == "loops":
        text = "int f(int x) {\n" + "".join(
            f"for (int j=0; j<x; ++j) {{ x += {i}; }}\n" for i in range(count)
        ) + "return x; }\n"
    elif kind == "deep":
        text = "int f(int x) {\n" + "{ int x=1;\n" * count + "return x;\n" + "}\n" * count + "}\n"
    elif kind == "symbols":
        prefix = "common_identifier_prefix_"
        text = "int f(int x) {\n" + "".join(
            f"int {prefix}{i}=x;\n" for i in range(count)
        ) + "".join(f"x += {prefix}{i};\n" for i in range(count)) + "return x; }\n"
    elif kind == "constants":
        text = "".join(f"constexpr int k{i} = {i} + 1;\n" for i in range(count))
    elif kind == "generic":
        text = "int f(int x) {\n" + "x += _Generic(x, int:1, default:0);\n" * count + "return x; }\n"
    elif kind == "types":
        # A linear-size DAG, not an exponentially expanded source spelling.
        text = "typedef int (*T0)(int);\n" + "".join(
            f"typedef T{i-1} (*T{i})(T{i-1},T{i-1});\n" for i in range(1, count + 1)
        ) + f"T{count} f(T{count} a);\nT{count} f(T{count} a);\n"
    elif kind == "diagnostics":
        text = "".join(
            f"int f{i}(void) {{ {{ int local=1; }} return local; }}\n" for i in range(count)
        )
    else:
        raise ValueError(f"unknown case: {kind}")
    return text, 1 if kind == "diagnostics" else 0


def digest(data):
    return hashlib.sha256(data).hexdigest()


def invoke(compiler, path, expected, objects, allocator, timeout):
    output = path.with_suffix(".o")
    output.unlink(missing_ok=True)
    command = [str(compiler), "cc", "-g0", f"-fregister-allocator={allocator}"]
    command += ["-c", "-o", str(output)] if objects else ["-fsyntax-only"]
    command.append(str(path))
    cpu_before = resource.getrusage(resource.RUSAGE_CHILDREN) if resource else None
    # GNU time provides per-child RSS, unlike RUSAGE_CHILDREN.ru_maxrss,
    # which is the maximum over all previous children and must not be diffed.
    with tempfile.TemporaryDirectory(prefix="buster-semantic-time-") as temporary:
        metrics = Path(temporary) / "usage"
        measured = command
        has_rss = platform.system() == "Linux" and Path("/usr/bin/time").is_file()
        if has_rss:
            measured = ["/usr/bin/time", "-f", "%M", "-o", str(metrics), "--"] + command
        start = time.perf_counter_ns()
        process = subprocess.Popen(measured, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                   start_new_session=os.name == "posix")
        try:
            stdout, stderr = process.communicate(timeout=timeout)
        except subprocess.TimeoutExpired:
            # Kill the compiler too, not just the optional GNU time wrapper.
            if os.name == "posix":
                os.killpg(process.pid, signal.SIGKILL)
            else:
                process.kill()
            process.communicate()
            raise
        completed = subprocess.CompletedProcess(command, process.returncode, stdout, stderr)
        wall_ns = time.perf_counter_ns() - start
        cpu_after = resource.getrusage(resource.RUSAGE_CHILDREN) if resource else None
        rss = int(metrics.read_text().splitlines()[-1]) if has_rss else None
    object_hash = digest(output.read_bytes()) if output.exists() else None
    output.unlink(missing_ok=True)
    if completed.returncode != expected:
        raise RuntimeError(f"{command!r}: expected status {expected}, got {completed.returncode}\n"
                           + completed.stderr.decode(errors="replace"))
    if objects and expected == 0 and object_hash is None:
        raise RuntimeError(f"successful compiler produced no object: {command!r}")
    cpu_ns = None
    if resource:
        cpu_ns = int(1e9 * (cpu_after.ru_utime + cpu_after.ru_stime - cpu_before.ru_utime - cpu_before.ru_stime))
    return {"command": command, "wall_ns": wall_ns, "cpu_ns": cpu_ns, "peak_rss_kib": rss,
            "status": completed.returncode, "stdout_sha256": digest(completed.stdout),
            "stderr_sha256": digest(completed.stderr), "object_sha256": object_hash}


def csv_positive(text):
    values = [int(value) for value in text.split(",")]
    if not values or any(value < 1 for value in values):
        raise argparse.ArgumentTypeError("expected comma-separated positive integers")
    return values


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--baseline", type=Path)
    parser.add_argument("--candidate", type=Path)
    parser.add_argument("--output", type=Path, default=Path("build/semantic-stress"))
    parser.add_argument("--sizes", type=csv_positive, default=[128, 512, 2048])
    parser.add_argument("--type-depths", type=csv_positive, default=[4, 8, 12])
    parser.add_argument("--kinds", default=",".join(KINDS) + ",types")
    parser.add_argument("--repetitions", type=int, default=7)
    parser.add_argument("--timeout", type=float, default=30)
    parser.add_argument("--allocator", choices=("none", "mir-stack", "fast", "quality"), default="none")
    parser.add_argument("--objects", action="store_true")
    parser.add_argument("--generate-only", action="store_true")
    parser.add_argument("--self-test", action="store_true")
    args = parser.parse_args()
    if args.self_test:
        for kind in KINDS + ("types",):
            small, expected = source(kind, 2)
            large, _ = source(kind, 4)
            assert small == source(kind, 2)[0] and len(large) > len(small)
            assert expected == (1 if kind == "diagnostics" else 0)
        print("7/7 deterministic source generators passed")
        return 0
    if args.repetitions < 1 or args.timeout <= 0:
        parser.error("repetitions and timeout must be positive")
    kinds = args.kinds.split(",")
    if any(kind not in KINDS + ("types",) for kind in kinds):
        parser.error("unknown stress kind")
    if not args.generate_only and (not args.baseline or not args.candidate):
        parser.error("provide --baseline and --candidate, or use --generate-only")
    args.output.mkdir(parents=True, exist_ok=True)
    compilers = {name: path.resolve() for name, path in (("baseline", args.baseline), ("candidate", args.candidate)) if path}
    report = {"platform": platform.platform(), "machine": platform.machine(), "python": platform.python_version(),
              "affinity": sorted(os.sched_getaffinity(0)) if hasattr(os, "sched_getaffinity") else None,
              "compiler_sha256": {name: digest(path.read_bytes()) for name, path in compilers.items()}, "cases": []}
    report_path = args.output / "results.json"
    try:
        for kind in kinds:
            for size in args.type_depths if kind == "types" else args.sizes:
                text, expected = source(kind, size)
                path = (args.output / f"{kind}-{size}.c").resolve()
                path.write_text(text, encoding="utf-8")
                if args.generate_only:
                    continue
                row = {"kind": kind, "size": size, "bytes": len(text.encode()), "source_sha256": digest(text.encode()), "samples": []}
                report["cases"].append(row)
                identity = None
                for iteration in range(args.repetitions + 1):
                    order = ("baseline", "candidate") if iteration % 2 == 0 else ("candidate", "baseline")
                    for name in order:
                        sample = invoke(compilers[name], path, expected, args.objects, args.allocator, args.timeout)
                        current = tuple(sample[key] for key in ("status", "stdout_sha256", "stderr_sha256", "object_sha256"))
                        if identity is None:
                            identity = current
                        if current != identity:
                            raise RuntimeError(f"nondeterministic or changed output/diagnostics: {kind}-{size}, {name}")
                        if iteration:
                            row["samples"].append(dict(sample, compiler=name, iteration=iteration))
                row["summary"] = {}
                for name in compilers:
                    samples = [sample for sample in row["samples"] if sample["compiler"] == name]
                    row["summary"][name] = {key: statistics.median(sample[key] for sample in samples)
                                             for key in ("wall_ns", "cpu_ns", "peak_rss_kib") if samples[0][key] is not None}
                print(kind, size, json.dumps(row["summary"], sort_keys=True), flush=True)
                report_path.write_text(json.dumps(report, indent=2) + "\n")
    finally:
        if not args.generate_only:
            report_path.write_text(json.dumps(report, indent=2) + "\n")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
